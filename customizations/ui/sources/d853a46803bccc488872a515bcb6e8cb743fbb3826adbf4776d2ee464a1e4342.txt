		function matchScore(parts, tokens) {
			if (tokens.length === 0) return 0;
			const normalized = parts.map((part) => normalizeSearch(part));
			let score = 0;
			for (const token of tokens) {
				let tokenScore = 0;
				for (let index = 0; index < normalized.length; index += 1) {
					const value = normalized[index];
					if (!value.includes(token)) continue;
					const weight = index === 0 ? 8 : index === 1 ? 4 : 2;
					tokenScore = Math.max(tokenScore, value === token ? weight + 4 : value.startsWith(token) ? weight + 2 : weight);
				}
				if (tokenScore === 0) return 0;
				score += tokenScore;
			}
			return score;
		}

		async function readKnowledgeFile(file) {
			if (!file || !/\.(txt|md)$/i.test(file.name)) throw new Error(TEXT.fileUnsupported);
			if (file.size > MAX_TEXT_FILE_BYTES) throw new Error(TEXT.fileTooLarge);
			let bytes;
			try { bytes = await file.arrayBuffer(); } catch { throw new Error(TEXT.fileReadFailed); }
			if (bytes.byteLength > MAX_TEXT_FILE_BYTES) throw new Error(TEXT.fileTooLarge);
			let body;
			try { body = new TextDecoder("utf-8", { fatal: true }).decode(bytes); } catch { throw new Error(TEXT.fileEncoding); }
			if (/[\u0000-\u0008\u000b\u000c\u000e-\u001f]/.test(body)) throw new Error(TEXT.fileEncoding);
			if (!body.trim()) throw new Error(TEXT.fileEmpty);
			return { title: clampText(file.name.replace(/\.(txt|md)$/i, ""), MAX_TITLE), body,
				source: { filename: file.name, format: /\.md$/i.test(file.name) ? "markdown" : "text", originalBytes: bytes.byteLength, importedAt: nowIso() } };
		}

		}

						body: "## 是什么\nQoder 知识中心（QMind）是一个本地知识库。你把资料存进去后，Qoder 在后续对话时能自动检索、引用这些资料来回答。\n\n## 怎么放入知识\n通过「添加来源」完成，支持三种方式：\n- 纯文本：直接粘贴一段内容。\n- HTTPS 链接：给出一个网址，系统抓取网页内容存入。\n- 本地文件：指定本机文件路径（PDF、代码文件、文档等）。\n\n放入时可以归到某个 Notebook（笔记本）下做分类管理。\n\n## 存入后有什么用\n- 跨会话记忆：知识持久化，新对话也能用到，不用每次重复贴资料。\n- 语义检索：不是简单的关键词匹配，而是理解意思后搜索。\n- 回答有据可查：Qoder 回答时会标注来自哪个来源，可追溯。\n\n## 后续怎么用\n- 自动触发：提问时 Qoder 自动从知识库检索相关内容作为上下文。\n- 主动检索：可以明确要求它在知识库里搜某个主题。\n- 浏览管理：列出所有 Notebook、列出所有来源、查看某个来源原文。\n\n## 特点\n- 本地存储，数据不上传。\n- 支持网页、文件、文本多种导入方式。\n- Notebook 分类组织。\n- 语义搜索，不死板。\n\n## 典型场景\n把项目的设计文档、API 文档、技术规范丢进去；之后问 Qoder 任何项目相关问题，它都能基于已存资料精准回答。",
						tags: ["Qoder", "知识中心", "QMind", "说明"],
						createdAt: nowIso(),
						updatedAt: nowIso()
					}
				]
			};
		}

		function KnowledgeApp(props) {
			return h(PersistenceGate, { component: KnowledgeContent, componentProps: props });
		}
		function KnowledgeContent(props) {
			const [data, setData, storageStatus] = usePortableState(KB_KEY);
			const [dbData, setDbData] = useState(() => readJson(DB_KEY, { version: SCHEMA, fields: [], records: [] }));
			const [query, setQuery] = useState("");
			const [tagFilter, setTagFilter] = useState("");
			const [editingId, setEditingId] = useState(null);
			const [confirmDel, setConfirmDel] = useState(null);
			const [title, setTitle] = useState("");
			const [body, setBody] = useState("");
			const [tags, setTags] = useState("");
			const [sort, setSort] = useState("updated");
			const [pendingImport, setPendingImport] = useState(null);
			const [fileReading, setFileReading] = useState(false);
			const fileInputRef = useRef(null);
			const fileReadEpoch = useRef(0);
			const [notice, setNotice] = useState("");
			const [collectionOpen, setCollectionOpen] = useState(false);
			const [collectionId, setCollectionId] = useState("");
			const [collectionDraft, setCollectionDraft] = useState(null);
			const [draftEntry, setDraftEntry] = useState(null);
			const [entryCollectionId, setEntryCollectionId] = useState("");
			const [workspaceFilter, setWorkspaceFilter] = useState("");
			const workspaceState = useSource(workbenchRuntime.workspaces?.list);
			const workspaces = (workspaceState?.items || []).map((item) => ({ ...item, id: item.workspaceId }));
			const noticeTimer = useRef(null);
			const searchRef = useRef(null);
			const showNotice = (text) => {
				setNotice(text);
				if (noticeTimer.current) clearTimeout(noticeTimer.current);
				noticeTimer.current = setTimeout(() => setNotice(""), 2800);
			};
			useEffect(() => () => {
				if (noticeTimer.current) clearTimeout(noticeTimer.current);
				fileReadEpoch.current++;
			}, []);
			useEffect(() => {
				const syncDatabase = (event) => {
					if (event && event.detail && event.detail.key !== DB_KEY) return;
					setDbData(readJson(DB_KEY, { version: SCHEMA, fields: [], records: [] }));
				};
				const focusSearch = (event) => {
					if ((event.ctrlKey || event.metaKey) && String(event.key).toLowerCase() === "k") {
						event.preventDefault();
						if (searchRef.current) searchRef.current.focus();
					}
				};
				window.addEventListener(STORE_EVENT, syncDatabase);
				window.addEventListener("keydown", focusSearch);
				return () => {
					window.removeEventListener(STORE_EVENT, syncDatabase);
					window.removeEventListener("keydown", focusSearch);
				};
			}, []);
			const seededRef = useRef("");
			useEffect(() => {
				const entry = data.entries.find((item) => item.id === editingId);
				if (entry && seededRef.current !== entry.id) {
					seededRef.current = entry.id;
					setTitle(entry.title || "");
					setBody(entry.body || "");
					setTags((entry.tags || []).join("、"));
					setEntryCollectionId(entry.collectionId);
				}
				if (!entry) seededRef.current = "";
			}, [editingId]);

			const allTags = useMemo(() => {
				const set = new Set();
				for (const entry of data.entries) for (const tag of entry.tags || []) set.add(tag);
				return [...set].sort((a, b) => a.localeCompare(b, "zh-Hans-CN"));
			}, [data]);
			const visible = useMemo(() => {
				const q = query.trim().toLowerCase();
				return data.entries
					.filter((entry) => !collectionId || entry.collectionId === collectionId)
					.filter((entry) => !workspaceFilter || data.collections.some((item) => item.id === entry.collectionId && (!item.workspaceId || item.workspaceId === workspaceFilter)))
					.filter((entry) => (tagFilter === "" || (entry.tags || []).includes(tagFilter)))
					.filter((entry) => q === "" || (entry.title || "").toLowerCase().includes(q) || (entry.body || "").toLowerCase().includes(q) || (entry.tags || []).some((tag) => tag.toLowerCase().includes(q)))
					.sort((a, b) => {
						if (sort === "title") return (a.title || "").localeCompare(b.title || "", "zh-Hans-CN");
						const field = sort === "created" ? "createdAt" : "updatedAt";
						return (b[field] || "").localeCompare(a[field] || "");
					});
			}, [data, query, tagFilter, sort, collectionId, workspaceFilter]);
			const tokens = useMemo(() => searchTokens(query), [query]);
			const globalResults = useMemo(() => {
				if (tokens.length === 0) return [];
				const results = [];
				for (const entry of data.entries) {
					if (workspaceFilter && !data.collections.some((item) => item.id === entry.collectionId && (!item.workspaceId || item.workspaceId === workspaceFilter))) continue;
					if (tagFilter !== "" && !(entry.tags || []).includes(tagFilter)) continue;
					const score = matchScore([entry.title, (entry.tags || []).join(" "), entry.body], tokens);
					if (score > 0) results.push({ kind: "knowledge", id: entry.id, score, title: entry.title || "（无标题）", body: searchSnippet(entry.body, tokens, 160), meta: (entry.tags || []).join(" · ") });
				}
				for (const record of dbData.records || []) {
					const values = (dbData.fields || []).map((field) => ({ field, value: record.cells && record.cells[field.id] != null ? String(record.cells[field.id]) : "" })).filter((item) => item.value.trim() !== "");
					const titlePart = values[0];
					const detail = values.map((item) => item.field.name + "：" + item.value).join(" · ");
					const score = matchScore([titlePart ? titlePart.value : "", (dbData.fields || []).map((field) => field.name).join(" "), detail], tokens);
					if (score > 0) results.push({ kind: "database", id: record.id, score, title: titlePart ? titlePart.value : "（空记录）", body: searchSnippet(detail, tokens, 180), meta: values.length + " 个已填字段" });
				}
				return results.sort((a, b) => b.score - a.score || a.title.localeCompare(b.title, "zh-Hans-CN")).slice(0, 100);
			}, [data, dbData, tokens, tagFilter, workspaceFilter]);
			const openDatabaseResult = (recordId) => {
				const detail = { tab: "database", recordId, query };
				try { sessionStorage.setItem(NAV_KEY, JSON.stringify(detail)); } catch (error) {}
				if (props.service) props.service.openTab({ type: "space-database" }, props.scope);
				setTimeout(() => window.dispatchEvent(new CustomEvent(NAV_EVENT, { detail })), 0);
			};

			const editing = data.entries.find((item) => item.id === editingId) || (draftEntry?.id === editingId ? draftEntry : null);
			const startNew = () => {
				const entry = { id: uid("kb"), title: "", body: "", tags: [], createdAt: nowIso(), updatedAt: nowIso() };
				setDraftEntry(entry);
				setTitle(""); setBody(""); setTags("");
				setEntryCollectionId(collectionId || data.collections[0].id);
				setCollectionOpen(true);
				setEditingId(entry.id);
			};
			const commit = async () => {
				if (!title.trim()) { showNotice(TEXT.titleRequired); return; }
				const latest = normalizeKnowledge(readJson(KB_KEY, data));
				if (!latest.collections.some((item) => item.id === entryCollectionId)) { showNotice(TEXT.collectionMissing); return; }
				if (draftEntry?.source && latest.entries.some((item) => item.collectionId === entryCollectionId && item.body === body)) { showNotice(TEXT.fileDuplicate); return; }
				const entry = { ...editing, title: clampText(title, MAX_TITLE), body: String(body || ""), tags: parseTags(tags), collectionId: entryCollectionId, updatedAt: nowIso() };
				if (await setData((prev) => ({ ...prev, entries: prev.entries.some((item) => item.id === editingId)
					? prev.entries.map((item) => item.id === editingId ? entry : item) : [entry, ...prev.entries] }))) {
					setDraftEntry(null); setEditingId(null);
				}
			};
			const addFile = async (event) => {
				const file = event.target.files?.[0];
				event.target.value = "";
				if (!file) return;
				const epoch = ++fileReadEpoch.current;
				setFileReading(true);
				try {
					const loaded = await readKnowledgeFile(file);
					if (epoch !== fileReadEpoch.current) return;
					const entry = { ...loaded, id: uid("kb"), tags: [], createdAt: nowIso(), updatedAt: nowIso() };
					setDraftEntry(entry); setTitle(entry.title); setBody(entry.body); setTags("");
					setEntryCollectionId(collectionId || data.collections[0].id);
					setCollectionOpen(true); setEditingId(entry.id); showNotice(TEXT.fileDraft);
				} catch (error) {
					if (epoch === fileReadEpoch.current) showNotice(error.message || TEXT.fileReadFailed);
				} finally { if (epoch === fileReadEpoch.current) setFileReading(false); }
			};
			const remove = async (id) => {
				if (await setData((prev) => ({ ...prev, entries: prev.entries.filter((item) => item.id !== id) })) && editingId === id) setEditingId(null);
			};
			const exportEntries = () => {
				if (downloadJson("knowledge-" + new Date().toISOString().slice(0, 10) + ".json", exportPayload("knowledge", data))) {
					showNotice("已导出 " + data.entries.length + " 条条目");
				}
			};
			const startImport = () => pickJsonFile((payload) => {
				const wrap = payload !== null && payload !== undefined && payload.app === "dsh-sidebar-spaces" ? payload : null;
				const space = wrap ? wrap.space : "knowledge";
				const part = wrap ? wrap.data : payload;
				if (!part || space !== "knowledge" || !Array.isArray(part.entries)) { showNotice("导入失败：不是有效的知识库备份文件"); return; }
				if (wrap && wrap.schema !== SCHEMA) { showNotice("导入失败：备份格式版本不受支持（schema " + wrap.schema + "）"); return; }
				if (part.version !== SCHEMA || part.entries.some((item) => !item || typeof item.id !== "string" || typeof item.title !== "string" || typeof item.body !== "string" || !Array.isArray(item.tags) || item.tags.some((tag) => typeof tag !== "string"))) { showNotice("导入失败：条目结构不正确"); return; }
				if (part.collections !== undefined && (!Array.isArray(part.collections) || part.collections.some((item) => !item || typeof item.id !== "string" || typeof item.title !== "string"))) { showNotice("导入失败：知识库结构不正确"); return; }
				const normalized = normalizeKnowledge(part);
				const incoming = [...new Map(normalized.entries.map((item) => [item.id, { ...item, title: clampText(item.title, MAX_TITLE), body: item.body, tags: parseTags(item.tags.join("、")), createdAt: typeof item.createdAt === "string" ? item.createdAt : nowIso(), updatedAt: typeof item.updatedAt === "string" ? item.updatedAt : nowIso() }])).values()];
				const existing = new Set(data.entries.map((item) => item.id));
				const fresh = incoming.filter((item) => !existing.has(item.id));
				setPendingImport({ total: incoming.length, fresh, collections: normalized.collections });
			});

			const overview = !collectionOpen && tokens.length === 0;
			return h("div", { className: "dss-root dss-knowledge-root" + (overview ? " is-overview" : ""), inert: storageStatus === "saving" ? true : undefined },
				h(StorageStatus, { status: storageStatus, readyText: TEXT.knowledgeSaved }),
				overview ? h("div", { className: "dss-kb-hero" },
					h("div", { className: "dss-kb-heading" }, h("span", { className: "dss-kb-mark", "aria-hidden": true }, Icon.book(24)),
						h("div", null, h("h1", null, TEXT.knowledgeHeading), h("p", null, TEXT.knowledgeHint))),
					h("div", { className: "dss-kb-summary" },
						h("span", null, h("strong", null, data.collections.length), TEXT.knowledgeCount),
						h("span", null, h("strong", null, data.entries.length), TEXT.entryCount))) : null,
				h("header", { className: "dss-toolbar" },
					h("div", { className: "dss-search" },
						Icon.search(14),
						h("input", {
							ref: searchRef, className: "dss-search-input", value: query, placeholder: TEXT.globalSearchPlaceholder,
							title: TEXT.globalSearchHint,
							onChange: (event) => setQuery(event.target.value)
						}),
						query ? h("button", { type: "button", className: "dss-iconbtn", "aria-label": "清空搜索", onClick: () => setQuery("") }, Icon.close(12)) : null
					),
					overview ? h("select", { className: "dss-select dss-kb-scope", value: workspaceFilter, "aria-label": TEXT.workspaceLabel, onChange: (event) => setWorkspaceFilter(event.target.value) },
						h("option", { value: "" }, TEXT.allWorkspaces),
						workspaces.map((item) => h("option", { key: item.id, value: item.id }, item.title || item.path))) : null,
					h("div", { className: "dss-toolbar-actions" },
					collectionOpen && tokens.length === 0 ? h("button", { type: "button", className: "dss-btn is-ghost", onClick: () => setCollectionOpen(false) }, "知识库概览") : null,
					collectionId ? h("button", { type: "button", className: "dss-btn", onClick: () => setCollectionDraft({ ...data.collections.find((item) => item.id === collectionId) }) }, TEXT.collectionEdit) : null,
					h("select", { className: "dss-select dss-sort", value: sort, "aria-label": "排序方式", onChange: (event) => setSort(event.target.value) },
						h("option", { value: "updated" }, "最近更新"),
						h("option", { value: "created" }, "创建时间"),
						h("option", { value: "title" }, "标题")
					),
					h("button", { type: "button", className: "dss-btn", title: "导出 JSON 备份", onClick: exportEntries }, Icon.exportIcon(13), "导出"),
					h("button", { type: "button", className: "dss-btn", title: "从 JSON 备份导入（新增合并，不覆盖现有）", disabled: !!editing || fileReading, onClick: startImport }, Icon.importIcon(13), TEXT.importBackup),
					h("input", { ref: fileInputRef, type: "file", hidden: true, accept: ".txt,.md", onChange: addFile, "aria-label": TEXT.addFile }),
					h("button", { type: "button", className: "dss-btn", title: TEXT.fileHint, disabled: !!editing || fileReading, onClick: () => fileInputRef.current?.click() }, Icon.importIcon(13), fileReading ? TEXT.fileReading : TEXT.addFile),
					h("button", { type: "button", className: "dss-btn", disabled: !!editing || fileReading, onClick: startNew }, Icon.plus(13), "新建条目")
				)
				),
				notice ? h("div", { className: "dss-notice", role: "status" }, notice) : null,
				h("section", { className: "dss-kb-overview", "aria-label": "知识库概览" },
					h("div", { className: "dss-kb-cards" },
						h("button", { type: "button", className: "dss-kb-create", onClick: () => setCollectionDraft({ id: uid("collection"), title: "", workspaceId: workspaceFilter }) }, h("span", { className: "dss-kb-create-icon" }, Icon.plus(22)), h("strong", null, TEXT.collectionCreate), h("small", null, TEXT.collectionCreateHint)),
						data.collections.filter((item) => !workspaceFilter || !item.workspaceId || item.workspaceId === workspaceFilter).map((collection) => h("button", { key: collection.id, type: "button", className: "dss-kb-card", onClick: () => { setCollectionId(collection.id); setTagFilter(""); setCollectionOpen(true); } },
							h("span", { className: "dss-kb-card-icon" }, Icon.book(18)),
							h("strong", null, collection.title),
							h("span", null, data.entries.filter((item) => item.collectionId === collection.id).length + " " + TEXT.entryCount),
							h("small", null, collection.workspaceId ? (workspaces.find((item) => item.id === collection.workspaceId)?.title || collection.workspaceId) : TEXT.globalWorkspace),
							h("span", { className: "dss-kb-open", "aria-hidden": true }, TEXT.collectionOpen + "  ↗")
						))
				)
				),
				allTags.length > 0 ? h("div", { className: "dss-chips" },
					h("button", { type: "button", className: "dss-chip" + (tagFilter === "" ? " is-active" : ""), onClick: () => setTagFilter("") }, "全部"),
					allTags.map((tag) => h("button", {
						key: tag, type: "button",
						className: "dss-chip" + (tagFilter === tag ? " is-active" : ""),
						onClick: () => setTagFilter(tagFilter === tag ? "" : tag)
					}, tag))
				) : null,
				h("div", { className: "dss-scroll" },
					tokens.length > 0 ? h("div", { className: "dss-results-summary", role: "status" },
						"找到 " + globalResults.length + " 项 · ",
						TEXT.knowledgeSource + " " + globalResults.filter((item) => item.kind === "knowledge").length + " · ",
						TEXT.databaseSource + " " + globalResults.filter((item) => item.kind === "database").length
					) : null,
					(tokens.length > 0 ? globalResults : visible).length === 0
						? h("div", { className: "dss-empty" }, tokens.length > 0 ? TEXT.noGlobalResults : tagFilter ? "没有匹配的条目" : "还没有条目，点「新建条目」开始")
						: (tokens.length > 0 ? globalResults : visible).map((entry) => h("article", {
							key: (entry.kind || "knowledge") + ":" + entry.id, className: "dss-item" + (entry.kind === "database" ? " is-database" : ""), tabIndex: 0, role: "button",
							onClick: () => entry.kind === "database" ? openDatabaseResult(entry.id) : setEditingId(entry.id),
							onKeyDown: (event) => {
								if (event.key === "Enter" || event.key === " ") {
									event.preventDefault();
									if (entry.kind === "database") openDatabaseResult(entry.id); else setEditingId(entry.id);
								}
							}
						},
							tokens.length > 0 ? h("div", { className: "dss-source-badge" }, entry.kind === "database" ? TEXT.databaseSource : TEXT.knowledgeSource) : null,
							h("div", { className: "dss-item-title" }, entry.title || "（无标题）"),
							entry.body ? h("div", { className: "dss-item-body" }, entry.body) : null,
							h("div", { className: "dss-item-meta" },
								tokens.length === 0 && (entry.tags || []).length > 0 ? h("span", { className: "dss-item-tags" }, entry.tags.map((tag, index) => h("span", { key: index, className: "dss-tag" }, tag))) : h("span", { className: "dss-item-time" }, entry.meta || ""),
								h("span", { className: "dss-item-time" }, tokens.length === 0 ? (entry.updatedAt || "").slice(0, 10) : "匹配度 " + entry.score)
							)
						))
				),
				editing ? h("div", { className: "dss-drawer" },
					h("div", { className: "dss-drawer-head" },
						h("span", { className: "dss-drawer-title" }, editing.title ? "编辑条目" : "新条目"),
						h("button", { type: "button", className: "dss-btn is-ghost", "aria-label": "关闭", onClick: () => setEditingId(null) }, Icon.close(13))
					),
					h("div", { className: "dss-field" },
						h("label", { className: "dss-label", htmlFor: "dss-entry-collection" }, TEXT.collectionMove),
						h("select", { id: "dss-entry-collection", className: "dss-select", value: entryCollectionId, onChange: (event) => setEntryCollectionId(event.target.value) }, data.collections.map((item) => h("option", { key: item.id, value: item.id }, item.title)))
					),
					h("div", { className: "dss-field" },
						h("label", { className: "dss-label", htmlFor: "dss-kb-title" }, "标题"),
						h("input", { id: "dss-kb-title", className: "dss-input", value: title, autoFocus: true, maxLength: MAX_TITLE, onChange: (event) => setTitle(event.target.value) })
					),
					h("div", { className: "dss-field" },
						h("label", { className: "dss-label", htmlFor: "dss-kb-body" }, "正文"),
						h("textarea", { id: "dss-kb-body", className: "dss-textarea", value: body, onChange: (event) => setBody(event.target.value), placeholder: "记录知识、结论、链接…" }),
						h("small", null, TEXT.fileBodyCount + body.length),
						editing.source ? h("p", { className: "dss-item-meta" }, TEXT.fileSource + "：" + editing.source.filename + " · " + editing.source.originalBytes + " bytes") : null
					),
					h("div", { className: "dss-field" },
						h("label", { className: "dss-label", htmlFor: "dss-kb-tags" }, "标签（顿号分隔，最多 " + MAX_TAGS + " 个）"),
						h("input", { id: "dss-kb-tags", className: "dss-input", value: tags, onChange: (event) => setTags(event.target.value), placeholder: "如：DSA、算法" })
					),
					h("div", { className: "dss-drawer-actions" },
						h("button", { type: "button", className: "dss-btn is-danger", onClick: () => setConfirmDel(editing.id) }, "删除条目"),
						h("button", { type: "button", className: "dss-btn", onClick: commit }, "完成")
					)
				) : null,
				collectionDraft ? h("div", { className: "dss-scrim" }, h("form", { className: "dss-dialog", role: "dialog", "aria-label": TEXT.collectionEdit, onSubmit: async (event) => {
					event.preventDefault();
					if (!collectionDraft.title.trim()) { showNotice(TEXT.collectionRequired); return; }
					const item = { ...collectionDraft, title: clampText(collectionDraft.title, MAX_TITLE) };
					if (await setData((prev) => ({ ...prev, collections: prev.collections.some((row) => row.id === item.id)
						? prev.collections.map((row) => row.id === item.id ? item : row) : [...prev.collections, item] }))) setCollectionDraft(null);
					} },
					h("h2", { className: "dss-kb-dialog-title" }, data.collections.some((item) => item.id === collectionDraft.id) ? TEXT.collectionEdit : TEXT.collectionCreate),
					h("label", { className: "dss-label", htmlFor: "dss-collection-title" }, TEXT.collectionName),

		function ChartCanvas(props) {
			const ref = useRef(null);
			const instRef = useRef(null);
			useEffect(() => {
				if (ref.current === null || typeof echarts === "undefined" || echarts === null) return;
				const chart = echarts.init(ref.current, null, { renderer: "canvas" });
				instRef.current = chart;
				chart.setOption(props.option);
				let ro = null;
				let onResize = null;
				if (typeof ResizeObserver !== "undefined") {
					ro = new ResizeObserver(() => chart.resize());
					ro.observe(ref.current);
				} else {
					onResize = () => chart.resize();
					window.addEventListener("resize", onResize);
				}
				return () => {
					if (ro) ro.disconnect();
					if (onResize) window.removeEventListener("resize", onResize);
					chart.dispose();
					instRef.current = null;
				};
			}, [props.option]);
			const savePng = () => {
				const chart = instRef.current;
				if (!chart || ref.current === null) return;
				try {
					const root = ref.current.closest(".dss-root");

		function ChartPanel(props) {
			const fields = props.fields || [];
			const records = props.records || [];
			const kind = props.kind || "bar";
			const field = fields.find((item) => item.id === props.fieldId) || fields[0];
			const measure = fields.find((item) => item.id === props.measureId && item.type === "number");
			const aggregate = props.aggregate || "sum";
			const limit = Math.max(5, Math.min(30, Number(props.limit) || 10));
			const option = useMemo(() => {
				if (field === undefined) return null;
				const rawValues = records.map((record) => record.cells && record.cells[field.id]).filter((value) => value !== undefined && value !== null && String(value).trim() !== "");
				const { ink, muted, accent, line, panel } = themeColors();
				const fontFamily = "'Segoe UI Variable Text','Segoe UI','Microsoft YaHei UI','PingFang SC',sans-serif";
				const statsBase = { total: records.length, populated: rawValues.length, unique: new Set(rawValues.map(String)).size };
				const base = { animationDuration: 350, textStyle: { fontFamily, color: muted }, grid: { left: 10, right: 18, top: 28, bottom: 10, containLabel: true } };
				if (field.type === "number") {
					const nums = rawValues.map(Number).filter((n) => Number.isFinite(n));
					if (nums.length === 0) return { __empty: "该字段暂无数字数据" };
					const min = Math.min(...nums);
					const max = Math.max(...nums);
					const sum = nums.reduce((a, b) => a + b, 0);
					const stats = Object.assign({}, statsBase, { count: nums.length, sum, avg: sum / nums.length, min, max });
					if (kind === "scatter" && measure && measure.id !== field.id) {
						const points = records.map((record) => [Number(record.cells && record.cells[field.id]), Number(record.cells && record.cells[measure.id])]).filter((pair) => pair.every(Number.isFinite));
						if (points.length === 0) return { __empty: "两个数字字段之间暂无成对数据" };
						return Object.assign({}, base, { stats, hint: field.name + " × " + measure.name + " · " + points.length + " 个有效点", tooltip: { trigger: "item", formatter: (p) => field.name + "：" + fmtNum(p.value[0]) + "<br/>" + measure.name + "：" + fmtNum(p.value[1]) }, xAxis: { type: "value", name: field.name, nameTextStyle: { color: muted }, axisLabel: { color: muted }, splitLine: { lineStyle: { color: line } } }, yAxis: { type: "value", name: measure.name, nameTextStyle: { color: muted }, axisLabel: { color: muted }, splitLine: { lineStyle: { color: line } } }, series: [{ type: "scatter", data: points, symbolSize: 9, itemStyle: { color: accent, opacity: 0.78 } }] });
					}
					const buckets = Math.max(5, Math.min(14, Math.ceil(Math.sqrt(nums.length))));
					const span = max - min || 1;
					const counts = new Array(buckets).fill(0);
					for (const n of nums) counts[Math.min(buckets - 1, Math.floor(((n - min) / span) * buckets))] += 1;
					const labels = counts.map((_, index) => fmtNum(min + (span / buckets) * index));
					return Object.assign({}, base, { stats, hint: "自动分为 " + buckets + " 个区间", tooltip: { trigger: "axis", axisPointer: { type: "shadow" }, formatter: (params) => { const p = params[0]; return labels[p.dataIndex] + " ~ " + fmtNum(min + (span / buckets) * (p.dataIndex + 1)) + "<br/>数量：" + p.value; } }, xAxis: { type: "category", data: labels, axisLabel: { color: muted, interval: "auto", fontSize: 10 }, axisLine: { lineStyle: { color: line } } }, yAxis: { type: "value", minInterval: 1, axisLabel: { color: muted }, splitLine: { lineStyle: { color: line } } }, series: [{ type: "bar", data: counts, barMaxWidth: 26, itemStyle: { color: accent, borderRadius: [4, 4, 0, 0] } }] });
				}
				const byMonth = field.type === "date" && new Set(rawValues.map((value) => String(value).slice(0, 10))).size > 60;
				const groups = new Map();
				for (const record of records) {
					const raw = record.cells && record.cells[field.id];
					if (raw === undefined || raw === null || String(raw).trim() === "") continue;
					const key = field.type === "date" ? String(raw).slice(0, byMonth ? 7 : 10) : String(raw);
					const group = groups.get(key) || { count: 0, values: [] };
					group.count += 1;
					if (measure) { const n = Number(record.cells && record.cells[measure.id]); if (Number.isFinite(n)) group.values.push(n); }
					groups.set(key, group);
				}
				const valueOf = (group) => {
					if (!measure) return group.count;
					if (group.values.length === 0) return 0;
					if (aggregate === "avg") return group.values.reduce((a, b) => a + b, 0) / group.values.length;
					if (aggregate === "min") return Math.min(...group.values);
					if (aggregate === "max") return Math.max(...group.values);
					return group.values.reduce((a, b) => a + b, 0);
				};
				let entries = [...groups.entries()].map(([key, group]) => [key, valueOf(group)]);
				if (entries.length === 0) return { __empty: "该字段暂无数据" };
				if (field.type === "date") entries.sort((a, b) => a[0].localeCompare(b[0])); else entries.sort((a, b) => b[1] - a[1]);
				if (field.type !== "date" && entries.length > limit) {
					const rest = entries.slice(limit).reduce((sum, entry) => sum + entry[1], 0);
					entries = entries.slice(0, limit);
					if (rest > 0 && (!measure || aggregate === "sum")) entries.push(["其他", rest]);
				}
				const metric = measure ? ({ sum: "合计", avg: "平均", min: "最小", max: "最大" }[aggregate] + " · " + measure.name) : "记录数";
				if (kind === "pie") return Object.assign({}, base, { stats: statsBase, hint: metric + " · 显示前 " + limit + " 项", color: PALETTE, tooltip: { trigger: "item", formatter: (p) => p.name + "：" + fmtNum(p.value) + "（" + p.percent + "%）" }, series: [{ type: "pie", radius: ["42%", "70%"], center: ["50%", "54%"], data: entries.map((entry) => ({ name: entry[0], value: entry[1] })), label: { color: muted, fontSize: 11, formatter: "{b} {d}%" }, labelLine: { lineStyle: { color: line } }, itemStyle: { borderColor: panel, borderWidth: 1 } }] });
				if (field.type === "date") return Object.assign({}, base, { stats: statsBase, hint: (byMonth ? "按月聚合" : "按日聚合") + " · " + metric, tooltip: { trigger: "axis" }, dataZoom: entries.length > 18 ? [{ type: "inside", start: 0, end: Math.max(20, Math.round(1800 / entries.length)) }] : undefined, xAxis: { type: "category", data: entries.map((entry) => entry[0]), axisLabel: { color: muted, hideOverlap: true, fontSize: 10 }, axisLine: { lineStyle: { color: line } } }, yAxis: { type: "value", minInterval: measure ? undefined : 1, axisLabel: { color: muted }, splitLine: { lineStyle: { color: line } } }, series: [kind === "line" ? { type: "line", data: entries.map((entry) => entry[1]), smooth: true, symbolSize: 5, lineStyle: { color: accent, width: 2 }, itemStyle: { color: accent }, areaStyle: { color: accent, opacity: 0.1 } } : { type: "bar", data: entries.map((entry) => entry[1]), barMaxWidth: 22, itemStyle: { color: accent, borderRadius: [4, 4, 0, 0] } }] });
				const labels = entries.map((entry) => entry[0]).reverse();
				const values = entries.map((entry) => entry[1]).reverse();
				return Object.assign({}, base, { stats: statsBase, hint: metric + " · 显示前 " + limit + " 项", tooltip: { trigger: "item", formatter: (p) => p.name + "：" + fmtNum(p.value) }, grid: { left: 10, right: 44, top: 10, bottom: 10, containLabel: true }, xAxis: { type: "value", minInterval: measure ? undefined : 1, axisLabel: { color: muted }, splitLine: { lineStyle: { color: line } } }, yAxis: { type: "category", data: labels, axisLabel: { color: muted, width: 100, overflow: "truncate", fontSize: 11 }, axisLine: { show: false }, axisTick: { show: false } }, series: [{ type: "bar", data: values, barMaxWidth: 18, itemStyle: { color: accent, borderRadius: [0, 4, 4, 0] }, label: { show: true, position: "right", color: ink, fontSize: 11, formatter: (p) => fmtNum(p.value) } }] });
			}, [field, records, kind, measure, aggregate, limit]);

			if (field === undefined) return h("div", { className: "dss-scroll dss-chartview" }, h("div", { className: "dss-empty" }, "先在表格视图里添加字段，再来图表看分布"));
			if (option === null || option.__empty) return h("div", { className: "dss-scroll dss-chartview" }, h("div", { className: "dss-chart-title" }, "「" + field.name + "」分析 · 共 " + records.length + " 条记录"), h("div", { className: "dss-empty" }, option ? option.__empty : "暂无数据"));
			const stats = option.stats;
			return h("div", { className: "dss-scroll dss-chartview" },
				h("div", { className: "dss-chart-title" }, "「" + field.name + "」分析 · 共 " + records.length + " 条记录"),
				stats ? h("div", { className: "dss-stats" },
					statChip("总记录", stats.total), statChip("已填", stats.populated), statChip("唯一值", stats.unique), statChip("完整率", stats.total ? Math.round(stats.populated / stats.total * 100) + "%" : "—"),
					stats.count !== undefined ? statChip("合计", fmtNum(stats.sum)) : null, stats.count !== undefined ? statChip("平均", fmtNum(stats.avg)) : null, stats.count !== undefined ? statChip("最小", fmtNum(stats.min)) : null, stats.count !== undefined ? statChip("最大", fmtNum(stats.max)) : null) : null,
				h(ChartCanvas, { option, name: field.name, height: kind === "scatter" ? 300 : 270 }),
				option.hint ? h("div", { className: "dss-chart-hint" }, option.hint) : null
			);
		}

		/** 总览统计：每个字段的填充率、唯一值数与数值/日期边界（纯函数，供测试）。 */
		function overviewStats(fields, records) {
			const total = records.length;
			return fields.map((field) => {
				let filled = 0;
				const uniq = new Set();
				let min = null;
				let max = null;
				for (const record of records) {
					const value = record.cells ? record.cells[field.id] : record[field.id];
					const empty = value === undefined || value === null || value === "";
					if (!empty) {
						filled += 1;
						uniq.add(String(value));
						const num = field.type === "number" ? Number(value) : null;
						if (num !== null && Number.isFinite(num)) {
							if (min === null || num < min) min = num;
							if (max === null || num > max) max = num;
						}
						if (field.type === "date" && (min === null || String(value) < min)) min = String(value);
						if (field.type === "date" && (max === null || String(value) > max)) max = String(value);
					}
				}
				return { id: field.id, name: field.name, type: field.type, total, filled, fillRate: total === 0 ? 0 : Math.round((filled / total) * 100), unique: uniq.size, min, max };
			});
		}

		async function inventorCall(action) {
			const response = await fetch("/sidebar-spaces/api/inventor/" + action, {
				method: "POST"
			});
			let payload = null;
			try { payload = await response.json(); } catch (error) {}
			if (!response.ok || !payload || payload.ok !== true) {
				const detail = payload && payload.error && payload.error.message ? payload.error.message : "Inventor 连接请求失败（HTTP " + response.status + "）";
				throw new Error(detail);
			}
			return payload.value || {};
		}

		function taskSearchResults(snapshot, contentHits, query, allowedIds) {
			const phrase = normalizeSearch(query);
			if (!phrase) return [];
			const hits = new Map(contentHits.map((item) => [item.sessionId, item.snippet]));
			const results = [];
			for (const id of snapshot?.ids || []) {
				const row = snapshot.byId[id];
				if (!row || row.blank || (allowedIds && !allowedIds.has(id))) continue;
				const title = row.displayTitle || row.title || id;
				const titleHit = normalizeSearch(title).includes(phrase);
				const bodyHit = hits.has(id);
				if (titleHit || bodyHit) results.push({ id, title, body: hits.get(id) || "", cwd: row.cwd || "",
					titleHit, bodyHit, updatedAt: row.updatedAt });
			}
			return results.sort((a, b) => Number(b.titleHit) - Number(a.titleHit) || b.updatedAt - a.updatedAt);
		}
		function startTaskContentSearch(service, query, publish, delay = 250) {
			const controller = new AbortController();
			const base = { query, items: [], hasMore: false };
			if (!query || !service) {
				publish({ ...base, status: service ? "idle" : "unavailable" });
				return () => controller.abort();
			}
			publish({ ...base, status: "loading" });
			const timer = setTimeout(async () => {
				try {
					const result = await service.search(query, controller.signal);
					if (controller.signal.aborted) return;
					publish(result.ok ? { query, ...result.value, status: "ready" } : { ...base, status: "error" });
				} catch (error) {
					if (!controller.signal.aborted) publish({ ...base, status: "error" });
				}
			}, delay);
			return () => { controller.abort(); clearTimeout(timer); };
		}

			}, [service, scope, tab.id]);
			return null;
		}
		function retireAutomationTabs(service) {
			let closing = false;
			const sync = () => {
				if (closing) return;
				const state = service.getSnapshot().state;
				const ids = new Set();
				const visit = node => {
					if (!node) return;
					if (node.kind === "leaf") for (const tab of node.tabs) { if (tab.type === "space-automation") ids.add(tab.id); }
					else if (node.kind === "split") node.children.forEach(visit);
				};
				visit(state?.splits); visit(state?.bottomSplits);
				for (const entry of state?.floats || []) if (entry.tab.type === "space-automation") ids.add(entry.tab.id);
				closing = true;
				try { for (const id of ids) service.closeTab(id); } finally { closing = false; }
			};
			const off = service.subscribeState(sync);
			sync();
			return off;
		}

		function normalizeExtensionCatalog(body) {
			const registry = body?.registry;
			if (!registry || !Array.isArray(registry.plugins) || !registry.categories || typeof registry.categories !== "object") throw new Error("invalid-extension-catalog");
			const rows = registry.plugins.map((row) => {
				if (!row || typeof row.name !== "string" || typeof row.url !== "string" || typeof row.owner !== "string") throw new Error("invalid-extension-row");
				const url = new URL(row.url);
				if (url.protocol !== "https:" || url.username || url.password) throw new Error("invalid-extension-url");
				return { ...row, category: (Array.isArray(row.category) ? row.category : [row.category]).filter(value => typeof value === "string"),
					description: typeof row.description?.zh === "string" ? row.description.zh : typeof row.description?.en === "string" ? row.description.en : "",
					npm: typeof row.npm === "string" ? row.npm : null, added: typeof row.added === "string" ? row.added : "" };
			});
			const categories = Object.fromEntries(Object.entries(registry.categories).map(([key, value]) => [key, {
				zh: typeof value?.zh === "string" ? value.zh : key,
				en: typeof value?.en === "string" ? value.en : key,
			}]));
			return { rows, categories };
		}
		function filterExtensions(rows, group, category, query, sort) {
			const words = query.trim().toLocaleLowerCase().split(/\s+/).filter(Boolean);
			return rows.filter(row => (group !== "skill" || row.category.includes("skill"))
				&& (group !== "connections" || row.category.some(key => ["identity", "notify", "remote"].includes(key)))
				&& (category === "all" || row.category.includes(category))
				&& words.every(word => `${row.name} ${row.owner} ${row.npm || ""} ${row.description}`.toLocaleLowerCase().includes(word)))
				.sort((a, b) => (sort === "latest" ? b.added.localeCompare(a.added) : 0) || a.name.localeCompare(b.name));
		}
		async function readExtensionData(path, signal) {
			const response = await fetch(path, { cache: "no-store", credentials: "same-origin", signal: AbortSignal.any([signal, AbortSignal.timeout(20000)]) });
			if (!response.ok) throw new Error(`extension-http-${response.status}`);
			return response.json();
		}

					if (!controller.signal.aborted) { setCatalog(value); setPhase("ready"); }

		function openExtensionManager(tab, onMissing, onSelected, query = "") {
			let tabSelected = false;
			return openSettingsSection("market", onMissing, onSelected, () => {
				if (!tabSelected) {
				const pattern = tab === "installed" ? /^(已安装|Installed)(\s*\(\d+\))?$/i : /^(发现|Discover)$/i;
				const button = Array.from(document.querySelectorAll('[role="dialog"] button')).find(item => !item.disabled && pattern.test(item.textContent.trim()));
				if (!button) return false;
				tabSelected = true; button.click();
				if (query) return false; // Wait for the selected tab's search input to mount.
				}
				if (query) {
					const input = Array.from(document.querySelectorAll('[role="dialog"] input')).find(item =>
					["搜索插件，例如：通知、终端、记忆…", "Search plugins: notify, terminal, memory…"].includes(item.placeholder));
					if (!input) return false;
					Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, "value").set.call(input, query);
					input.dispatchEvent(new Event("input", { bubbles: true }));
				}
				return true;
			});
		}

		function openStandaloneWorkbench(type) {
			if (!standaloneHost) {
				standaloneReturnFocus = document.activeElement;
				standaloneHost = document.createElement("div");
				standaloneHost.className = "dss-standalone-host";
				document.body.appendChild(standaloneHost);
				standaloneRoot = createRoot(standaloneHost);
			}
			standaloneHost.classList.toggle("dss-extension-host", type === "space-extensions");
			standaloneLayoutCleanup?.();
			standaloneLayoutCleanup = void 0;
			standaloneHost.style.removeProperty("left");
			{
				// Slot wrappers may use display:contents (zero-sized rect); measure the visible aside.
				const sidebar = document.querySelector('aside.dcu-root, [data-slot="sidebar"] aside');
				if (sidebar) {
					const align = () => { if (standaloneHost) standaloneHost.style.left = Math.max(0, sidebar.getBoundingClientRect().right) + "px"; };
					const observer = new ResizeObserver(align);
					observer.observe(sidebar); align();
					standaloneLayoutCleanup = () => observer.disconnect();
				}
			}
			standaloneRoot.render(h(StandaloneWorkbench, { type, key: type }));
		}

		// The installed settings contract owns open state; footer slots have no public
		// openSection callback. Keep this compatibility adapter scoped and cancellable.
		const SETTINGS_SECTIONS = Object.freeze({ plugins: ["插件", "Plugins"], market: ["插件市场", "Plugin Market"], automation: ["定时任务", "Scheduled tasks", "自动化"] });
		let cancelSettingsNavigation;
		function openSettingsSection(section, onMissing, onSelected = () => {}, followUp) {
			cancelSettingsNavigation?.();
			const labels = SETTINGS_SECTIONS[section];
			let finished = false;
			let sectionSelected = false;
			const cleanup = () => {
				finished = true; observer.disconnect(); clearTimeout(timer);
				if (cancelSettingsNavigation === cleanup) cancelSettingsNavigation = void 0;
			};
			const select = () => {
				if (finished) return false;
				if (!sectionSelected) {
				const target = Array.from(document.querySelectorAll('[role="dialog"] nav button')).find((button) =>
					!button.disabled && labels?.includes(button.textContent.replace(/\s+/g, " ").trim()));
				if (!target) return false;
				sectionSelected = true; target.click();
				}
				if (followUp && !followUp()) return false;
				cleanup(); onSelected(); return true;
			};
			const observer = new MutationObserver(select);
			// A manager handoff also waits for the native remote catalog (15 s host timeout).
			const timer = setTimeout(() => { if (!finished && !select()) { cleanup(); onMissing(); } }, followUp ? 25000 : 2500);
			cancelSettingsNavigation = cleanup;
			observer.observe(document.body, { childList: true, subtree: true });
			if (select()) return cleanup;
			// Do not open Settings over another form or guess a partially matching label.
			if (!document.querySelector('[role="dialog"]')) {
				const launcher = Array.from(document.querySelectorAll('button[aria-haspopup="dialog"]')).find((button) =>
					!button.disabled && [button.textContent, button.getAttribute("aria-label"), button.getAttribute("title")]
						.some(value => /^(设置|settings)$/i.test((value || "").trim())));
				if (!launcher) { cleanup(); onMissing(); return cleanup; }
				launcher.click(); select();
			}
			return cleanup;
		}

		function fail(phase, error) {
			console.error("[dsh-sidebar-spaces] " + phase + " error:", error);
			try {
				const bar = document.createElement("div");
				bar.style.cssText = "position:fixed;left:8px;bottom:8px;z-index:2147483000;max-width:70vw;padding:8px 12px;font:12px/1.5 ui-monospace,SFMono-Regular,Menlo,monospace;color:#f2a1a1;background:#1b1b22;border:1px solid #f2a1a1;border-radius:8px;white-space:pre-wrap";
				bar.textContent = "[dsh-sidebar-spaces] " + phase + " error: " + (error instanceof Error ? error.message : String(error));
				document.body.appendChild(bar);
			} catch (barError) {}
		}

		function withBetterSidebar(ctx, mount) {
			const fiber = ctx.inject(["betterSidebar"], (inner) => {
				inner.effect(() => mount(inner.betterSidebar), "dsh-sidebar-spaces: available sidebar");
			});
			return () => fiber.dispose();
		}

		// The 0.1.5 official panel bypasses workspaces.openPath. Coordinate
		// presentation, not file routing: both providers retain their own tabs.
		function coordinateRightPanels(native, cards, observe) {
			if (typeof cards.setPanelOpen !== "function") return () => {};
			let disposed = false;
			let wasOpen = cards.getSnapshot().state?.panelOpen === true;
			const syncNative = () => {
				if (!disposed && native.isExpanded() && cards.getSnapshot().state?.panelOpen) cards.setPanelOpen(false);
			};
			const off = cards.subscribeState(() => {
				const open = cards.getSnapshot().state?.panelOpen === true;
				const opening = open && !wasOpen;
				wasOpen = open;
				if (!disposed && opening && native.isExpanded()) native.toggleExpanded();
			});
			// Restored sessions can initially have both providers expanded.
			const offNative = observe(syncNative);
			syncNative();
			return () => {
				disposed = true;
				off();
				offNative();
			};
		}
		function observePanelPresentation(sync) {
			// The official controller has no public subscription. Its own tab
			// actions bypass the outer service facade. After a rendered commit,
			// read isExpanded(); never infer state from hashed CSS class names.
			let frame = 0;
			const observer = new MutationObserver(() => {
				if (!frame) frame = requestAnimationFrame(() => { frame = 0; sync(); });
			});
			observer.observe(document.getElementById("root") || document.body, { subtree: true, childList: true, attributes: true, attributeFilter: ["data-sidebar-right-open", "data-sidebar-right-panel"] });
			return () => { observer.disconnect(); if (frame) cancelAnimationFrame(frame); };
		}

				}, "dsh-sidebar-spaces: optional right panel coordination");
