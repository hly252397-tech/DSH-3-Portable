    function renderRollbackCapsule(value){
      const capsule=document.getElementById('rollback-capsule');if(!capsule)return;
      const update=value&&value.status,rolled=Boolean(update)&&String(update.kind)==='rolled-back';
      capsule.dataset.visible=String(rolled);
      if(!rolled){capsule.firstElementChild.textContent='';capsule.removeAttribute('title');capsule.removeAttribute('aria-label');return}
      const zh=shellZh(),current=typeof value.currentVersion==='string'&&value.currentVersion?value.currentVersion:'';
      const reverted=typeof update.version==='string'&&update.version?update.version:'';
      const text=reverted?(zh?`已回退：${reverted} → ${current||'未知'}`:`Rolled back: ${reverted} → ${current||'unknown'}`):(zh?'已回退到旧版本':'Rolled back to an older version');
      capsule.firstElementChild.textContent='⚠ '+text;
      const detail=typeof update.detail==='string'&&update.detail?update.detail:(typeof update.message==='string'&&update.message?update.message:text);
      capsule.title=detail;capsule.setAttribute('aria-label',detail);
    }
    function renderDesktopUpdate(value){
      if(!value||!value.status)return;const button=document.getElementById('settings-btn'),update=value.status,phase=String(update.kind||'idle');
      const busy=['checking','downloading','verifying','building','deploying','validating'].includes(phase);const hasProgress=typeof update.overallProgress==='number'&&Number.isFinite(update.overallProgress);const overall=hasProgress?Math.max(0,Math.min(100,update.overallProgress)):0;
      const indicator=!busy?'none':phase==='downloading'&&hasProgress?'progress':'spinner';
      button.dataset.phase=phase;button.dataset.busy=String(busy);button.dataset.indicator=indicator;button.style.setProperty('--update-progress',String(indicator==='progress'?overall/100:0));button.disabled=false;
      const zh=shellZh();const labels={idle:zh?'检查更新':'Check for updates',checking:zh?'正在检查更新':'Checking for updates',none:zh?'当前已是最新版本':'Up to date',available:zh?'发现新版本，点击查看':'Update available; click to review',downloading:zh?'正在下载':'Downloading',verifying:zh?'正在验证':'Verifying',building:zh?'正在构建候选':'Building candidate',ready:zh?'候选已就绪，点击部署':'Candidate ready; click to deploy',deploying:zh?'正在部署':'Deploying',validating:zh?'正在启动验证':'Validating startup',completed:zh?'更新已完成':'Update completed',['rolled-back']:zh?'更新失败，已自动回滚':'Update failed and rolled back',error:zh?'更新失败，点击查看':'Update failed; click to review'};
      const detail=typeof update.detail==='string'&&update.detail?update.detail:labels[phase]||labels.idle;const title=busy?detail+(hasProgress?' · '+Math.round(overall)+'%':''):detail;button.title=button.ariaLabel=(zh?'DSH 设置':'DSH settings')+' · '+title;
      renderRollbackCapsule(value);
    }

    document.addEventListener('pointerdown',event=>{if(openMenu&&event.target!==openMenu)clearOpenMenu()},true);window.addEventListener('blur',()=>clearOpenMenu());
    document.querySelectorAll('[data-action]').forEach(button=>button.addEventListener('click',()=>api.action(button.dataset.action)));
    document.querySelector('.bar').addEventListener('pointerdown',event=>{const button=event.target.closest('.bar .nav');if(!button||button.disabled)return;button.classList.remove('press');void button.offsetWidth;button.classList.add('press')});

    const restartBtn=document.getElementById('restart-btn');
    function cancelRestartConfirm(){if(!restartConfirming)return;restartConfirming=false;clearTimeout(restartConfirmTimer);restartBtn.classList.remove('confirming');restartBtn.setAttribute('aria-pressed','false');restartBtn.title=restartBtn.ariaLabel=shellZh()?'重启应用':'Restart App';renderStatus()}
    restartBtn.addEventListener('click',async()=>{
      if(!restartConfirming){
        restartConfirming=true;restartBtn.classList.add('confirming');restartBtn.setAttribute('aria-pressed','true');renderStatus();
        restartBtn.title=restartBtn.ariaLabel=shellZh()?'再次点击确认重启（Esc 取消）':'Click again to confirm restart (Esc to cancel)';

      }else{
        clearTimeout(restartConfirmTimer);restartConfirming=false;restartExecuting=true;
        restartBtn.classList.remove('confirming');restartBtn.classList.add('spinning');restartBtn.setAttribute('aria-pressed','false');renderStatus();
        restartBtn.title=restartBtn.ariaLabel=shellZh()?'正在重启…':'Restarting…';restartBtn.disabled=true;
        await api.action('app-restart');
      }
    });
    document.addEventListener('keydown',e=>{if(e.key==='Escape')cancelRestartConfirm()});

    api.getDesktopUpdateState().then(renderDesktopUpdate);api.onDesktopUpdateState(renderDesktopUpdate);
